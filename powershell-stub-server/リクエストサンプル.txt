$body = @"
<soap:Envelope xmlns:soap="http://www.w3.org/2003/05/soap-envelope" xmlns:tem="http://tempurl.org/">
    <soap:Header/>
    <soap:Body>
        <tem:Search>
            <tem:Request>
                <data>TEST</data>
            </tem:Request>
        </tem:Search>
    </soap:Body>
</soap:Envelope>
"@

$headers = @{
    "Content-Type" = "text/xml;charset=UTF-8"
    "SOAPAction"   = '"http://tempurl.org/Search"'
}

$response = Invoke-WebRequest -Uri "http://localhost:8080/SampleService.asmx" `
-Method Post `
-Headers $headers `
-Body $body

$response.Content >> \Desktop\response.xml
